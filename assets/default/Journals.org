* Journals
#+BEGIN_SRC holon_sql :id journals::src::0
SELECT b.*, 1 AS expand_default FROM block b JOIN block_tags bt ON bt.block_id = b.id AND bt.tag = 'Page' WHERE b.parent_id = 'block:journals' ORDER BY b.content DESC
#+END_SRC
#+BEGIN_SRC render :id journals::render::0
list(#{sortkey: "-content", item_template: column(render_entity(), divider())})
#+END_SRC
