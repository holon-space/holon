#+BEGIN_SRC holon_sql :id left_sidebar::src::0
SELECT b.* FROM block b JOIN block_tags bt ON bt.block_id = b.id WHERE bt.tag = 'Page' AND b.id != 'block:__default__'

** Main Panel
:PROPERTIES:
:ID: default-main-panel
:END:
#+BEGIN_SRC holon_gql :id default-main-panel::src::0
MATCH (fr:focus_root), (root:block)<-[:CHILD_OF*0..20]-(d:block) WHERE fr.region = 'main' AND root.id = fr.root_id RETURN d
#+END_SRC

#+BEGIN_SRC holon_gql :id default-right-sidebar::src::0
