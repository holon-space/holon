#+BEGIN_SRC holon_sql :id default-main-panel::src::0
SELECT root.*
FROM focus_roots fr
JOIN block root ON root.id = fr.root_id
JOIN navigation_cursor nc ON nc.region = fr.region AND nc.history_id = fr.history_id
WHERE fr.region = 'main'
